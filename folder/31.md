```python
# platform/publishers/github_provider.py
import os
import subprocess
from github import Github, GithubException
from .base import GitHostProvider, RepositoryTarget

class GitHubProvider(GitHostProvider):
    def __init__(self, token: str):
        self.client = Github(token)

    async def ensure_repository_exists(self, target: RepositoryTarget) -> str:
        user = self.client.get_user()
        try:
            # Check if repo exists under the target owner
            if target.owner == user.login:
                repo = user.create_repo(target.repo_name, private=target.is_private)
            else:
                org = self.client.get_organization(target.owner)
                repo = org.create_repo(target.repo_name, private=target.is_private)
            return repo.clone_url
        except GithubException as e:
            if e.status == 422: # Already exists
                return f"https://github.com/{target.owner}/{target.repo_name}.git"
            raise

    async def push_bundle(self, target: RepositoryTarget, bundle_path: str, isr_lineage: list[str]) -> None:
        # Execute git commands locally in the bundle_path
        # Configuring author dynamically based on the target routing
        subprocess.run(["git", "config", "user.name", target.commit_author.name], cwd=bundle_path, check=True)
        subprocess.run(["git", "config", "user.email", target.commit_author.email], cwd=bundle_path, check=True)
        
        subprocess.run(["git", "init", "-b", "main"], cwd=bundle_path, check=True)
        subprocess.run(["git", "add", "."], cwd=bundle_path, check=True)
        
        # Semantic commit based on ISR lineage
        commit_msg = f"chore: initial generation from ISR\n\nLineage: {', '.join(isr_lineage)}"
        subprocess.run(["git", "commit", "-m", commit_msg], cwd=bundle_path, check=True)
        
        remote_url = await self.ensure_repository_exists(target)
        subprocess.run(["git", "remote", "add", "origin", remote_url], cwd=bundle_path, check=True)
        subprocess.run(["git", "push", "-u", "origin", "main"], cwd=bundle_path, check=True)

    async def configure_repository_settings(self, target: RepositoryTarget) -> None:
```

```python
        self.provider = provider

    def _evaluate_success_criteria(self, evidence: CertificationEvidence) -> bool:
        """Strictly evaluates the Exit Gates before any publishing occurs."""
        if evidence.test_pass_rate < self.settings.min_test_pass_rate: return False
        if evidence.cyclomatic_complexity > self.settings.max_cyclomatic_complexity: return False
        if evidence.security_vulnerabilities > self.settings.max_security_vulnerabilities: return False
        if self.settings.require_lint_pass and not evidence.lint_passed: return False
        return True

    def _determine_target_owner(self, auth_context: Optional[AuthContext]) -> str:
        """
        Primary: User's GitHub.
        Fallback: kimiti4 (if user is not authenticated or lacks repo creation scopes).
        """
        if auth_context and auth_context.github_username and auth_context.has_repo_scope:
            return auth_context.github_username
        return self.settings.fallback_owner

    def _determine_author(self, target_owner: str) -> GitCommitAuthor:
        if target_owner == self.settings.fallback_owner:
            return GitCommitAuthor(name="Tiannara Evolution Engine", email=self.settings.fallback_email)
        # Default to the user's context if pushing to their own account
        return GitCommitAuthor(name="Tiannara Evolution Engine", email="noreply@tiannara.local")

    async def process_and_publish(self, evidence: CertificationEvidence, bundle_path: str, auth_context: Optional[AuthContext] = None):
        # 1. GATE: Do not proceed if success criteria are not met
        if not self._evaluate_success_criteria(evidence):
            print(f"[GATE FAILED] Project {evidence.project_id} did not meet exit criteria. Archiving locally.")
            # Move to quarantine directory instead of pushing
            return 

        # 2. ROUTE: Determine where to push
        target_owner = self._determine_target_owner(auth_context)
        author = self._determine_author(target_owner)
        
        target = RepositoryTarget(
            owner=target_owner,
            repo_name=evidence.project_id,
            is_private=True, # Default to private for generated calibration projects
            commit_author=author
        )

        # 3. PUBLISH
        print(f"[GATE PASSED] Routing {evidence.project_id} to {target_owner}/{target.repo_name}")
        await self.provider.push_bundle(target, bundle_path, evidence.isr_lineage)
        await self.provider.configure_repository_settings(target)
```

```python
from pydantic import BaseModel, Field


class FitnessVector(BaseModel):
    """Multi-objective fitness. Never collapse to a single aggregate score."""

    metrics: dict[str, float] = Field(default_factory=dict)

    def dominates(self, other: "FitnessVector") -> bool:
        """Pareto dominance: >= on all objectives, > on at least one."""
        keys = set(self.metrics) & set(other.metrics)
        if not keys:
            return False
        at_least_as_good = all(self.metrics[k] >= other.metrics[k] for k in keys)
        strictly_better = any(self.metrics[k] > other.metrics[k] for k in keys)
        return at_least_as_good and strictly_better
```

```python
import tempfile
import traceback
from dataclasses import dataclass

from tiannara.domain.models.bundle import SystemDeploymentBundle
from tiannara.domain.models.evidence import CertificationEvidence, Verdict
from tiannara.domain.models.fitness import FitnessVector
from tiannara.domain.ports import CompilerBackend, EvolutionEngine, IntentCompiler


@dataclass
class PipelineResult:
    evidence: CertificationEvidence
    bundle: SystemDeploymentBundle | None


class ExecutionPipeline:
    """Intent -> ISR -> Evolution -> Compilation.

    Verification and publication are separate constitutional concerns
    handled by the harness and publisher respectively.
    """

    def __init__(
        self,
        intent_compiler: IntentCompiler,
        evolution_engine: EvolutionEngine,
        backends: dict[str, CompilerBackend],
    ) -> None:
        self._intent_compiler = intent_compiler
        self._evolution_engine = evolution_engine
        self._backends = backends

    async def execute(
        self,
        project_id: str,
        statement: str,
        target_backend: str,
        hints: dict,
    ) -> PipelineResult:
        backend = self._backends.get(target_backend)
        if backend is None:
            raise ValueError(f"Unknown compiler backend: {target_backend}")

        try:
            isr = self._intent_compiler.compile(statement, hints)
            isr.system_id = project_id
            isr.lineage.append(f"intent:{project_id}")

            candidate = self._evolution_engine.evolve(isr)
            isr.lineage.append(f"genome:{candidate.genome.genome_id}")
            isr_hash = isr.content_hash()

            output_dir = tempfile.mkdtemp(prefix=f"tiannara-{project_id}-")
            bundle = backend.compile(isr, candidate.genome, output_dir)

            evidence = CertificationEvidence(
                project_id=project_id,
                isr_hash=isr_hash,
                genome_id=candidate.genome.genome_id,
                backend_name=backend.name,
                compilation_success=True,
                fitness=FitnessVector(metrics={"compilation": 1.0}),
            )
            return PipelineResult(evidence=evidence, bundle=bundle)

        except Exception as exc:  # Evidence of failure is still evidence.
            evidence = CertificationEvidence(
                project_id=project_id,
                isr_hash="unavailable",
                genome_id="unavailable",
                backend_name=target_backend,
                compilation_success=False,
                verdict=Verdict.FAIL,
                error=f"{type(exc).__name__}: {exc}\n{traceback.format_exc()}",
            )
            return PipelineResult(evidence=evidence, bundle=None)
```

## `application/publisher/gate_evaluator.py`

```python
from dataclasses import dataclass

from tiannara.domain.models.evidence import (
    CertificationEvidence,
    GateResult,
    GateStatus,
)


@dataclass(frozen=True)
class GatePolicy:
    """Declarative success criteria. Injected, never hardcoded."""

    min_test_pass_rate: float
    require_compilation: bool
    require_security_scan: bool
    max_security_vulnerabilities: int


class ExitGateEvaluator:
    def __init__(self, policy: GatePolicy) -> None:
        self._policy = policy

    def evaluate(self, evidence: CertificationEvidence) -> tuple[bool, list[GateResult]]:
        results: list[GateResult] = []

        if self._policy.require_compilation:
            results.append(
                GateResult(
                    gate="compilation",
                    status=GateStatus.PASS if evidence.compilation_success else GateStatus.FAIL,
                    detail="Compiled successfully" if evidence.compilation_success else "Compilation failed",
                )
            )

        if evidence.test_run is None:
            results.append(
                GateResult(gate="tests", status=GateStatus.NOT_EVALUATED, detail="No test run recorded")
            )
        else:
            ok = evidence.test_run.pass_rate >= self._policy.min_test_pass_rate
            results.append(
                GateResult(
                    gate="tests",
                    status=GateStatus.PASS if ok else GateStatus.FAIL,
                    detail=f"pass_rate={evidence.test_run.pass_rate:.3f} "
                           f"required>={self._policy.min_test_pass_rate}",
                )
            )

        if not evidence.security_scan_performed:
            status = GateStatus.FAIL if self._policy.require_security_scan else GateStatus.NOT_EVALUATED
            results.append(GateResult(gate="security", status=status, detail="No scan performed"))
        else:
            ok = evidence.security_vulnerabilities <= self._policy.max_security_vulnerabilities
            results.append(
                GateResult(
                    gate="security",
                    status=GateStatus.PASS if ok else GateStatus.FAIL,
                    detail=f"vulnerabilities={evidence.security_vulnerabilities}",
                )
            )

        passed = all(r.status != GateStatus.FAIL for r in results) and any(
            r.status == GateStatus.PASS for r in results
        )
        return passed, results
```

## `application/publisher/publisher_orchestrator.py`

```python
import logging
import shutil
from dataclasses import dataclass
from pathlib import Path

from tiannara.domain.models.bundle import SystemDeploymentBundle
from tiannara.domain.models.evidence import CertificationEvidence, Verdict
from tiannara.domain.ports import EvidenceLedger, RepositoryPublisher

from .gate_evaluator import ExitGateEvaluator

logger = logging.getLogger(__name__)


@dataclass(frozen=True)
class PublishingIdentity:
    """Routing identity. Resolved at runtime — never hardcoded."""

    owner: str
    author_name: str
    author_email: str


@dataclass(frozen=True)
class AuthContext:
    """Present when an end user authenticates with repo scope."""

    github_username: str
    has_repo_scope: bool


class PublisherOrchestrator:
    def __init__(
        self,
        publisher: RepositoryPublisher,
        gate_evaluator: ExitGateEvaluator,
        ledger: EvidenceLedger,
        fallback_identity: PublishingIdentity,
        quarantine_dir: Path,
    ) -> None:
        self._publisher = publisher
        self._gates = gate_evaluator
        self._ledger = ledger
        self._fallback = fallback_identity
        self._quarantine_dir = quarantine_dir
        self._quarantine_dir.mkdir(parents=True, exist_ok=True)

    def _resolve_identity(self, auth: AuthContext | None) -> PublishingIdentity:
        if auth and auth.has_repo_scope:
            return PublishingIdentity(
                owner=auth.github_username,
                author_name="Tiannara Evolution Engine",
                author_email="noreply@tiannara.local",
            )
        return self._fallback

    async def process(
        self,
        bundle: SystemDeploymentBundle | None,
        evidence: CertificationEvidence,
        auth: AuthContext | None = None,
    ) -> None:
        passed, gate_results = self._gates.evaluate(evidence)
        evidence.gate_results = gate_results
        evidence.verdict = Verdict.PASS if passed else Verdict.FAIL

        if not passed or bundle is None:
            self._quarantine(bundle, evidence)
        else:
            identity = self._resolve_identity(auth)
            url = await self._publisher.publish(
                bundle=bundle,
                evidence=evidence,
                owner=identity.owner,
                author_name=identity.author_name,
                author_email=identity.author_email,
            )
            logger.info("Published %s -> %s", evidence.project_id, url)

        # Every outcome is recorded. The ledger is the evolutionary memory.
        self._ledger.append(evidence)

    def _quarantine(self, bundle: SystemDeploymentBundle | None, evidence: CertificationEvidence) -> None:
        evidence.verdict = Verdict.QUARANTINED
        if bundle is not None and bundle.path.exists():
            destination = self._quarantine_dir / evidence.project_id
            if destination.exists():
                shutil.rmtree(destination)
            shutil.move(str(bundle.path), destination)
            logger.warning("Quarantined %s -> %s", evidence.project_id, destination)
        else:
            logger.warning("Quarantined %s (no bundle)", evidence.project_id)
```

## `application/harness/calibration_harness.py`

```python
import asyncio
import logging
import time

from tiannara.domain.models.fitness import FitnessVector
from tiannara.domain.ports import ExecutionEnvironment

from ..pipeline.execution_pipeline import ExecutionPipeline
from ..publisher.publisher_orchestrator import AuthContext, PublisherOrchestrator
from .manifest import StratifiedManifest

logger = logging.getLogger(__name__)


class StratifiedCalibrationHarness:
    """Runs the stratified project matrix with bounded concurrency.

    Single-node implementation. For multi-node scale-out, replace the
    semaphore loop with a workflow engine (Temporal/Argo) without
    touching the domain or application logic.
    """

    def __init__(
        self,
        pipeline: ExecutionPipeline,
        environment: ExecutionEnvironment,
        orchestrator: PublisherOrchestrator,
        max_concurrency: int = 5,
    ) -> None:
        self._pipeline = pipeline
        self._environment = environment
        self._orchestrator = orchestrator
        self._semaphore = asyncio.Semaphore(max_concurrency)

    async def run(self, manifest: StratifiedManifest, auth: AuthContext | None = None) -> None:
        logger.info("Stratification:\n%s", manifest.stratification_report())
        tasks = [self._run_project(entry.model_dump(), auth) for entry in manifest.projects]
        await asyncio.gather(*tasks, return_exceptions=False)

    async def _run_project(self, entry: dict, auth: AuthContext | None) -> None:
        async with self._semaphore:
            project_id = entry["id"]
            started = time.monotonic()
            logger.info("[%s] started", project_id)

            result = await self._pipeline.execute(
                project_id=project_id,
                statement=entry["intent"],
                target_backend=entry["target_backend"],
                hints={
                    "domain": entry["domain"],
                    "complexity_tier": entry["complexity_tier"],
                    "constraints": entry["constraints"],
                },
            )
            evidence = result.evidence

            if result.bundle is not None:
                try:
                    test_run = await self._environment.run_verification(result.bundle)
                    evidence.test_run = test_run
                    evidence.fitness = FitnessVector(
                        metrics={
                            "compilation": 1.0,
                            "test_pass_rate": test_run.pass_rate,
                            "verification_latency_s": round(time.monotonic() - started, 3),
                        }
                    )
                finally:
                    await self._environment.teardown(result.bundle)

            await self._orchestrator.process(result.bundle, evidence, auth)
            logger.info("[%s] finished verdict=%s", project_id, evidence.verdict.value)
```

```python
import asyncio
import logging
import os
import stat
import subprocess
import tempfile
from pathlib import Path

import httpx

from tiannara.domain.models.bundle import SystemDeploymentBundle
from tiannara.domain.models.evidence import CertificationEvidence

logger = logging.getLogger(__name__)


class GitHubRepositoryPublisher:
    """GitHub adapter behind the RepositoryPublisher port.

    Credentials are passed to git via GIT_ASKPASS so the token never
    appears in remote URLs, process arguments, or logs.
    """

    API = "https://api.github.com"

    def __init__(self, token: str, default_branch: str = "main") -> None:
        self._token = token
        self._default_branch = default_branch
        self._client = httpx.AsyncClient(
            base_url=self.API,
            headers={
                "Authorization": f"Bearer {token}",
                "Accept": "application/vnd.github+json",
            },
            timeout=30.0,
        )

    async def publish(
        self,
        bundle: SystemDeploymentBundle,
        evidence: CertificationEvidence,
        owner: str,
        author_name: str,
        author_email: str,
    ) -> str:
        repo_name = bundle.project_id
        await self._ensure_repository(owner, repo_name)

        remote_url = f"https://github.com/{owner}/{repo_name}.git"
        await asyncio.to_thread(
            self._git_push,
            bundle.path,
            remote_url,
            author_name,
            author_email,
            evidence,
        )
        await self._create_release(owner, repo_name, tag="v0.1.0", evidence=evidence)
        return remote_url

    async def _ensure_repository(self, owner: str, repo_name: str) -> None:
        resp = await self._client.post(
            "/user/repos",
            json={"name": repo_name, "private": True, "auto_init": False},
        )
        if resp.status_code == 422:  # already exists — idempotent
            return
        if resp.status_code == 404:  # owner may be an organization
            resp = await self._client.post(
                f"/orgs/{owner}/repos",
                json={"name": repo_name, "private": True, "auto_init": False},
            )
        resp.raise_for_status()

    def _git_push(
        self,
        bundle_dir: Path,
        remote_url: str,
        author_name: str,
        author_email: str,
        evidence: CertificationEvidence,
    ) -> None:
        askpass = self._write_askpass()
        env = {
            **os.environ,
            "GIT_ASKPASS": askpass,
            "TIANNARA_GIT_TOKEN": self._token,
            "GIT_TERMINAL_PROMPT": "0",
        }
        try:
            commands = [
                ["git", "init", "-b", self._default_branch],
                ["git", "config", "user.name", author_name],
                ["git", "config", "user.email", author_email],
                ["git", "add", "."],
                [
                    "git",
                    "commit",
                    "-m",
                    (
                        "chore: compiled artifact from evolved ISR\n\n"

            for cmd in commands:
                subprocess.run(cmd, cwd=bundle_dir, env=env, check=True,
                               capture_output=True, text=True)
        finally:
            os.unlink(askpass)

        with os.fdopen(fd, "w") as fh:
            fh.write('#!/bin/sh\necho "$TIANNARA_GIT_TOKEN"\n')
        os.chmod(path, os.stat(path).st_mode | stat.S_IEXEC)
        return path

    async def _create_release(self, owner: str, repo: str, tag: str,
                              evidence: CertificationEvidence) -> None:
```

```python
from tiannara.domain.models.bundle import SystemDeploymentBundle

def _sanitize(name: str) -> str:
    return re.sub(r"[^a-z0-9-]", "-", name.lower()).strip("-")


class DockerComposeEnvironment:
    """Ephemeral verification sandbox.

    Runs the bundle's `tests` service inside Docker Compose, parses the
    JSON summary emitted by the smoke test, then destroys everything.
    """

    def __init__(self, timeout_seconds: int = 300) -> None:
        self._timeout = timeout_seconds

    async def run_verification(self, bundle: SystemDeploymentBundle) -> TestRunResult:
        project = _sanitize(bundle.project_id)
        started = time.monotonic()
        proc = await asyncio.to_thread(
            subprocess.run,
            ["docker", "compose", "-p", project, "run", "--rm", "tests"],
            cwd=bundle.path,
            capture_output=True,
            text=True,
            timeout=self._timeout,
        )
        duration = time.monotonic() - started
        total, failed = self._parse_summary(proc.stdout)

        logs_path = bundle.path / "verification.log"
        logs_path.write_text(proc.stdout + "\n--- stderr ---\n" + proc.stderr)

        return TestRunResult(
            passed=proc.returncode == 0,
            exit_code=proc.returncode,
            total_tests=total,
            failed_tests=failed,
            duration_seconds=round(duration, 3),
            logs_path=str(logs_path),
        )

    async def teardown(self, bundle: SystemDeploymentBundle) -> None:
        project = _sanitize(bundle.project_id)
        await asyncio.to_thread(
            subprocess.run,
            ["docker", "compose", "-p", project, "down", "-v", "--remove-orphans"],
            cwd=bundle.path,
            capture_output=True,
            timeout=self._timeout,
        )
        shutil.rmtree(bundle.path, ignore_errors=True)

    @staticmethod
    def _parse_summary(stdout: str) -> tuple[int, int]:
        for line in reversed(stdout.splitlines()):
            line = line.strip()
            if line.startswith("TIANNARA_TEST_SUMMARY"):
                payload = json.loads(line.split("=", 1)[1])
                return int(payload["total"]), int(payload["failed"])
        return 0, 0
```

## `infrastructure/ledger/jsonl_evidence_ledger.py`

```python
import hashlib
import json
import threading
from pathlib import Path

from tiannara.domain.models.evidence import CertificationEvidence

GENESIS_HASH = "0" * 64


class JsonlEvidenceLedger:
    """Append-only, hash-chained evidence store.

    Every record embeds the hash of its predecessor, making the ledger
    tamper-evident. This is the substrate Phase 38 learns from.
    """

    def __init__(self, path: str) -> None:
        self._path = Path(path)
        self._path.parent.mkdir(parents=True, exist_ok=True)
        self._lock = threading.Lock()

    def append(self, evidence: CertificationEvidence) -> CertificationEvidence:
        with self._lock:
            previous = self._last_hash()
            evidence.previous_hash = previous
            evidence.record_hash = self._hash_record(evidence, previous)
            with self._path.open("a", encoding="utf-8") as fh:
                fh.write(evidence.model_dump_json() + "\n")
        return evidence

    def verify_chain(self) -> bool:
        if not self._path.exists():
            return True
        previous = GENESIS_HASH
        with self._path.open("r", encoding="utf-8") as fh:
            for line in fh:
                record = json.loads(line)
                expected = self._hash_record(
                    CertificationEvidence.model_validate(
                        {**record, "previous_hash": None, "record_hash": None}
                    ),
                    previous,
                )
                if record["previous_hash"] != previous or record["record_hash"] != expected:
                    return False
                previous = record["record_hash"]
        return True

    def _last_hash(self) -> str:
        if not self._path.exists():
            return GENESIS_HASH
        last_line = ""
        with self._path.open("r", encoding="utf-8") as fh:
            for line in fh:
                last_line = line
        if not last_line.strip():
            return GENESIS_HASH
        return json.loads(last_line)["record_hash"]

    @staticmethod
    def _hash_record(evidence: CertificationEvidence, previous: str) -> str:
        payload = evidence.model_dump_json(exclude={"previous_hash", "record_hash"})
        return hashlib.sha256(f"{previous}:{payload}".encode()).hexdigest()
```
