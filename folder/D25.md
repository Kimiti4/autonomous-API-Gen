class ActionFirewall:

def canonical_json(obj: Any) -> str:
    return json.dumps(
        obj,
        sort_keys=True,
        separators=(",", ":"),
        ensure_ascii=True,
    )


def canonical_bytes(obj: Any) -> bytes:
    return canonical_json(obj).encode("utf-8")


def sha256_bytes(data: bytes) -> str:
    return hashlib.sha256(data).hexdigest()


def sha256_obj(obj: Any) -> str:
    return sha256_bytes(canonical_bytes(obj))


def sha256_file(path: Path) -> str:
    return hashlib.sha256(path.read_bytes()).hexdigest()


def fail(message: str) -> None:
    raise FailClosed(message)


def normalize_ref(ref: Any) -> str:
    if not isinstance(ref, str):
        return ""
    if "#" in ref:
        return ref.split("#", 1)[1]
    return ref


def norm_tokens(values: Any) -> Set[str]:
    if values is None:
        return set()
    if isinstance(values, str):
        values = [values]
    if isinstance(values, dict):
        values = list(values.keys())
    if not isinstance(values, list):
        values = [values]
    return {str(x).strip().lower() for x in values if str(x).strip()}


def extract_obligations(value: Any) -> Set[str]:
    if isinstance(value, dict):
        return norm_tokens(list(value.keys()))
    if isinstance(value, list):
        return norm_tokens(value)
    if value is None:
        return set()
    return norm_tokens([value])


def load_and_hash_json(path: Path, label: str) -> tuple[Dict[str, Any], Set[str], str]:
    path = Path(path)
    if not path.is_file():
        raise FailClosed(f"{label} artifact missing: {path}")

    raw_sha = sha256_file(path).lower()
    try:
        data = json.loads(path.read_text(encoding="utf-8"))
    except Exception as exc:
        raise FailClosed(f"{label} artifact is not valid canonical JSON: {path}: {exc}")

    if not isinstance(data, dict):
        raise FailClosed(f"{label} artifact must contain a JSON object: {path}")

    logical_sha = sha256_obj(data).lower()
    return data, {raw_sha, logical_sha}, raw_sha


def ensure_hash(actual_hashes: Iterable[str], expected: Optional[str], label: str) -> str:
    if not expected:
        raise FailClosed(f"{label} expected hash missing")

    expected_norm = expected.lower()
    actual_norm = {h.lower() for h in actual_hashes}

    if expected_norm not in actual_norm:
        raise FailClosed(f"{label} hash mismatch: expected {expected_norm}")

    return expected_norm


def verify_objective_source(root: Path, source_path: str) -> Dict[str, Any]:
    path = root / normalize_ref(source_path)
    data, hashes, _ = load_and_hash_json(path, "D22 objective source")

    source_sha = ensure_hash(
        hashes,
        EXPECTED_OBJECTIVE_SOURCE_SHA256,
        "D22 objective source",
    )

    if data.get("objective_id") != EXPECTED_OBJECTIVE_ID:
        fail("objective source objective_id is not VS1-OBJ-001")

    return {
        "source_ref": normalize_ref(source_path),
        "source_sha256": source_sha,
        "objective_hash": sha256_obj(data),
    }


def verify_d22(
    root: Path,
    d22_path: str,
    objective_source: Dict[str, Any],
) -> Dict[str, Any]:
    path = root / normalize_ref(d22_path)
    data, hashes, raw_sha = load_and_hash_json(path, "D22 objective intake")

    status = str(data.get("status", "")).upper()
    if not any(token in status for token in ("PASS", "OBJECTIVE_ADMITTED")):
        fail("D22 status is not PASS / OBJECTIVE_ADMITTED")

    if data.get("objective_id") != EXPECTED_OBJECTIVE_ID:
        fail("D22 objective_id is not VS1-OBJ-001")

    source_ref = normalize_ref(data.get("source_ref") or objective_source["source_ref"])
    source_sha = str(data.get("source_sha256") or "").lower()

    if not source_ref:
        fail("D22 source_ref missing")
    if not source_sha:
        fail("D22 source_sha256 missing")

    if source_ref != objective_source["source_ref"]:
        source_data, source_hashes, _ = load_and_hash_json(
            root / source_ref,
            "D22 referenced objective source",
        )
        ensure_hash(source_hashes, source_sha, "D22 source_ref hash")
        ensure_hash(
            source_hashes,
            EXPECTED_OBJECTIVE_SOURCE_SHA256,
            "D22 objective source pinned hash",
        )
        objective_hash = sha256_obj(source_data)
    else:
        if source_sha != objective_source["source_sha256"]:
            fail("D22 source_sha256 does not match verified objective source hash")
        objective_hash = objective_source["objective_hash"]

    return {
        "d22_hash": sha256_obj(data),
        "d22_file_sha256": raw_sha,
        "objective_id": EXPECTED_OBJECTIVE_ID,
        "objective_hash": data.get("objective_hash") or objective_hash,
        "source_ref": source_ref,
        "source_sha256": source_sha,
    }


def verify_d23(root: Path, d23_path: str) -> Dict[str, Any]:
    path = root / normalize_ref(d23_path)
    data, _, raw_sha = load_and_hash_json(path, "D23 candidate generation")

    if data.get("objective_id") != EXPECTED_OBJECTIVE_ID:
        fail("D23 objective_id is not VS1-OBJ-001")

    evidence_value = str(
        data.get("generation_evidence")
        or data.get("evidence")
        or data.get("generation_hash")
        or ""
    ).lower()

    if not evidence_value.startswith(EXPECTED_D23_EVIDENCE_PREFIX.lower()):
        fail("D23 generation evidence does not match expected D23 evidence prefix")

    candidates = data.get("candidates")
    if isinstance(candidates, dict):
        normalized: List[Dict[str, Any]] = []
        for candidate_id, candidate in candidates.items():
            if isinstance(candidate, dict):
                item = dict(candidate)
                item.setdefault("candidate_id", candidate_id)
                normalized.append(item)
            else:
                normalized.append(
                    {
                        "candidate_id": candidate_id,
                        "candidate_hash": candidate,
                    }
                )
        candidates = normalized

    if not isinstance(candidates, list):
        fail("D23 candidates missing or malformed")

    selected = None
    for candidate in candidates:
        if not isinstance(candidate, dict):
            continue
        candidate_id = candidate.get("candidate_id") or candidate.get("id")
        if candidate_id == EXPECTED_SELECTED_CANDIDATE:
            selected = candidate
            break

    if selected is None:
        fail("D24-selected candidate is not present in D23 candidate generation")

    if not selected.get("candidate_hash"):
        fail("D23 selected candidate hash missing")

    return {
        "d23_hash": sha256_obj(data),
        "d23_file_sha256": raw_sha,
        "generation_evidence": evidence_value,
        "selected_candidate": selected,
    }


def verify_candidate_hash(
    root: Path,
    d24: Dict[str, Any],
    d23_selected: Dict[str, Any],
    candidate_hash: str,
) -> str:
    candidate_hash_norm = candidate_hash.lower()

    representations = []
    if d24.get("candidate_representation") is not None:
        representations.append(d24["candidate_representation"])
    if d23_selected.get("candidate_representation") is not None:
        representations.append(d23_selected["candidate_representation"])

    for representation in representations:
        if sha256_obj(representation).lower() == candidate_hash_norm:
            return "candidate_representation"

    artifact_refs = []
    if d24.get("candidate_artifact_path"):
        artifact_refs.append(
            (
                normalize_ref(d24.get("candidate_artifact_path")),
                d24.get("candidate_artifact_sha256"),
            )
        )

    if d23_selected.get("artifact_path"):
        artifact_refs.append(
            (
                normalize_ref(d23_selected.get("artifact_path")),
                d23_selected.get("artifact_sha256") or d23_selected.get("candidate_hash"),
            )
        )

    if d23_selected.get("candidate_artifact_path"):
        artifact_refs.append(
            (
                normalize_ref(d23_selected.get("candidate_artifact_path")),
                d23_selected.get("candidate_artifact_sha256")
                or d23_selected.get("candidate_hash"),
            )
        )

    for artifact_ref, expected_artifact_hash in artifact_refs:
        if not artifact_ref:
            continue
        _, artifact_hashes, _ = load_and_hash_json(root / artifact_ref, "candidate artifact")
        if candidate_hash_norm in artifact_hashes:
            if expected_artifact_hash:
                ensure_hash(artifact_hashes, str(expected_artifact_hash), "candidate artifact")
            return "candidate_artifact"

    fail("selected candidate hash could not be cryptographically verified")
    return ""


def verify_d24(
    root: Path,
    d24_path: str,
    d23: Dict[str, Any],
) -> Dict[str, Any]:
    path = root / normalize_ref(d24_path)
    data, _, raw_sha = load_and_hash_json(path, "D24 architecture selection")

    selected_candidate = data.get("selected_candidate") or data.get("candidate_id")
    if selected_candidate != EXPECTED_SELECTED_CANDIDATE:
        fail("D24 selected_candidate is not the authoritative D24 candidate")

    if data.get("objective_id") != EXPECTED_OBJECTIVE_ID:
        fail("D24 objective_id is not VS1-OBJ-001")

    selection_hash = str(data.get("selection_hash") or "").lower()
    candidate_hash = str(data.get("candidate_hash") or "").lower()

    if not selection_hash:
        fail("D24 selection_hash missing")
    if not candidate_hash:
        fail("D24 candidate_hash missing")

    selection_payload = data.get("selection_payload") or data.get("selection_inputs")
    if selection_payload is None:
        fail("D24 selection_payload missing; selection hash cannot be recomputed")

    computed_selection_hash = sha256_obj(selection_payload).lower()
    if computed_selection_hash != selection_hash:
        fail("D24 selection_hash does not match canonical selection_payload")

    architecture = data.get("architecture") or data.get("selected_architecture")
    if not isinstance(architecture, dict):
        fail("D24 architecture missing or malformed")

    d23_selected = d23.get("selected_candidate", {})
    d23_candidate_hash = str(d23_selected.get("candidate_hash") or "").lower()

    if d23_candidate_hash and d23_candidate_hash != candidate_hash:
        fail("D24 candidate_hash does not match D23 selected candidate hash")

    candidate_verified_by = verify_candidate_hash(
        root,
        data,
        d23_selected,
        candidate_hash,
    )

    return {
        "content": data,
        "architecture": architecture,
        "d24_hash": sha256_obj(data),
        "d24_file_sha256": raw_sha,
        "candidate_id": selected_candidate,
        "candidate_hash": candidate_hash,
        "selection_hash": selection_hash,
        "candidate_hash_verified_by": candidate_verified_by,
        "backend_id": data.get("backend_id") or architecture.get("backend_id") or "",
    }

def get_priority_values(architecture: Dict[str, Any]) -> List[str]:
    data_model = architecture.get("data_model", {})
    if not isinstance(data_model, dict):
        data_model = {}

    task = data_model.get("Task") or data_model.get("task") or {}
    if not isinstance(task, dict):
        task = {}

    fields = task.get("fields", {})
    if not isinstance(fields, dict):
        fields = {}

    priority = fields.get("priority") or fields.get("Priority") or {}
    if not isinstance(priority, dict):
        priority = {}

    values = priority.get("values") or priority.get("enum") or []

    if not values:
        domain = architecture.get("domain", {})
        if isinstance(domain, dict):
            priority_domain = domain.get("priority") or domain.get("Priority") or {}
            if isinstance(priority_domain, dict):
                values = priority_domain.get("values") or priority_domain.get("enum") or []

    if isinstance(values, str):
        values = [values]
    if not isinstance(values, list):
        values = []

    return [str(v).strip().upper() for v in values if str(v).strip()]

def collect_capabilities(architecture: Dict[str, Any]) -> Set[str]:
    capabilities: List[Any] = []

    top = architecture.get("objective_capabilities")
    if isinstance(top, dict):
        capabilities.extend(top.keys())
    elif isinstance(top, list):
        capabilities.extend(top)

    components = architecture.get("components", [])
    if isinstance(components, list):
        for component in components:
            if not isinstance(component, dict):
                continue
            component_caps = component.get("objective_capabilities", [])
            if isinstance(component_caps, dict):
                capabilities.extend(component_caps.keys())
            elif isinstance(component_caps, list):
                capabilities.extend(component_caps)

    return norm_tokens(capabilities)


def collect_obligations(architecture: Dict[str, Any]) -> Set[str]:
    obligations: Set[str] = set()

    component_mappings = architecture.get("component_mappings")
    if isinstance(component_mappings, dict):
        for value in component_mappings.values():
            obligations.update(extract_obligations(value))

    components = architecture.get("components", [])
    if isinstance(components, list):
        for component in components:
            if not isinstance(component, dict):
                continue
            obligations.update(
                extract_obligations(
                    component.get("isr_obligations")
                    or component.get("maps_to_isr")
                )
            )

    return obligations


def get_component_id(component: Any) -> str:
    if isinstance(component, dict):
        for key in ("id", "component_id", "name"):
            value = component.get(key)
            if value:
                return str(value)
        return ""
    return str(component)


def build_component_mappings(architecture: Dict[str, Any]) -> Dict[str, List[str]]:
    components = architecture.get("components", [])
    if not isinstance(components, list) or not components:
        fail("architecture components missing or empty")

    top_mappings = architecture.get("component_mappings")
    if not isinstance(top_mappings, dict):
        top_mappings = {}

    result: Dict[str, List[str]] = {}

    for component in components:
        component_id = get_component_id(component)
        if not component_id:
            fail("architecture component missing id")

        obligations: Set[str] = set()

        if isinstance(component, dict):
            obligations.update(
                extract_obligations(
                    component.get("isr_obligations")
                    or component.get("maps_to_isr")
                )
            )

        if not obligations and component_id in top_mappings:
            obligations.update(extract_obligations(top_mappings[component_id]))

        if not obligations:
            fail(f"architecture component has no ISR obligation mapping: {component_id}")

        result[component_id] = sorted(obligations)

    return result


def validate_architecture(architecture: Dict[str, Any]) -> Dict[str, str]:
    if not isinstance(architecture, dict):
        fail("selected architecture missing or malformed")

    boundary = architecture.get("authority_boundary")
    if not isinstance(boundary, dict):
        fail("architecture must contain an explicit machine-readable authority_boundary")

    false_values = {"FALSE", "NOT_PERFORMED", "NOT_AUTHORIZED", "NO"}
    for operation in sorted(PROHIBITED_OPERATIONS):
        value = boundary.get(operation, None)
        if value is False:
            continue
        if str(value).strip().upper() in false_values:
            continue
        fail(f"authority_boundary must explicitly deny prohibited operation: {operation}")

    capabilities = collect_capabilities(architecture)
    missing_capabilities = REQUIRED_CAPABILITIES - capabilities
    if missing_capabilities:
        fail(
            "architecture does not cover required objective capabilities: "
            + ", ".join(sorted(missing_capabilities))
        )

    security = architecture.get("security")
    if not isinstance(security, dict):
        fail("architecture must contain a security object")

    missing_security = {
        control for control in REQUIRED_SECURITY if not security.get(control)
    }
    if missing_security:
        fail(
            "architecture security controls missing or false: "
            + ", ".join(sorted(missing_security))
        )

    priority_values = set(get_priority_values(architecture))
    if priority_values != PRIORITY_VALUES:
        fail("priority domain must be exactly LOW, MEDIUM, HIGH")

    api = architecture.get("api")
    if not isinstance(api, dict):
        fail("architecture must contain an api object")

    api_norm = {str(k).lower(): v for k, v in api.items()}
    missing_ops = REQUIRED_API_OPS - set(api_norm.keys())
    if missing_ops:
        fail("architecture API missing required operations: " + ", ".join(sorted(missing_ops)))

    if not api.get("preserve_existing_crud") and not api_norm.get("preserve_existing_crud"):
        fail("architecture API must explicitly preserve existing CRUD behavior")

    if "priority" not in api_field_names(api_norm.get("create"), "accepts"):
        fail("API create must accept priority")
    if "priority" not in api_field_names(api_norm.get("update"), "accepts"):
        fail("API update must accept priority")
    if "priority" not in api_field_names(api_norm.get("retrieve"), "returns"):
        fail("API retrieve must return priority")

    filter_fields = (
        api_field_names(api_norm.get("filter"), "query")
        | api_field_names(api_norm.get("filter"), "accepts")
    )
    if "priority" not in filter_fields:
        fail("API filter must accept priority")

    frontend = architecture.get("frontend")
    if not isinstance(frontend, dict):
        fail("architecture must contain a frontend object")
    if not frontend.get("display_priority"):
        fail("frontend must display task priority")
    if not frontend.get("filter_by_priority"):
        fail("frontend must support filtering by priority")

    persistence = architecture.get("persistence")
    if not isinstance(persistence, dict):
        fail("architecture must contain a persistence object")
    if not persistence.get("priority_persisted"):
        fail("persistence must persist priority")
    if not persistence.get("restart_durability"):
        fail("persistence must survive restart")
    if persistence.get("destructive_migration"):
        fail("destructive migration is not authorized")

    migration = persistence.get("migration")
    if isinstance(migration, dict) and migration.get("destructive"):
        fail("destructive migration is not authorized")

    events = architecture.get("events")
    if not isinstance(events, dict):
        fail("architecture must contain an events object")
    if not events.get("preserve_existing_event_contracts"):
        fail("architecture must preserve existing event contracts")
    if events.get("new_event_contracts"):
        fail("new event contracts are not authorized unless explicitly represented by D24")

    obligations = collect_obligations(architecture)
    missing_obligations = REQUIRED_ISR_OBLIGATIONS - obligations
    if missing_obligations:
        fail(
            "architecture does not cover required ISR obligations: "
            + ", ".join(sorted(missing_obligations))
        )

    component_mappings = build_component_mappings(architecture)

    coverage = {
        "architecture": f"{len(REQUIRED_ISR_OBLIGATIONS & obligations)}/{len(REQUIRED_ISR_OBLIGATIONS)}",
        "requirements": f"{len(REQUIRED_CAPABILITIES)}/{len(REQUIRED_CAPABILITIES)}",
        "security": f"{len(REQUIRED_SECURITY)}/{len(REQUIRED_SECURITY)}",
        "apis": "5/5",
        "data_models": "1/1",
        "events": "1/1",
        "components": f"{len(component_mappings)}/{len(component_mappings)}",
    }

    return coverage


def compile_implementation_plan(

    d22: Dict[str, Any],
    d23: Dict[str, Any],

    firewall.authorize("compile_implementation")

    }


def compute_implementation_identity(
    plan: Dict[str, Any],
    d22: Dict[str, Any],
    d23: Dict[str, Any],
    d24: Dict[str, Any],
    objective_source: Dict[str, Any],
    isr: Dict[str, Any],
) -> tuple[str, str]:
    payload = {
        "plan": plan,
        "isr_sha256": isr["isr_sha256"],
        "objective_id": EXPECTED_OBJECTIVE_ID,
        "objective_hash": d22["objective_hash"],
        "objective_source_sha256": objective_source["source_sha256"],
        "candidate_id": d24["candidate_id"],
        "candidate_hash": d24["candidate_hash"],
        "selection_hash": d24["selection_hash"],
        "d22_hash": d22["d22_hash"],
        "d23_hash": d23["d23_hash"],
        "d24_hash": d24["d24_hash"],
        "policy_version": POLICY_VERSION,
    }

    implementation_hash = sha256_obj(payload).lower()
    implementation_id = f"vs1-impl-obj001-{implementation_hash[:16]}"
    return implementation_hash, implementation_id


def verify_behavior_evidence(path: Path, implementation_hash: str) -> Dict[str, Any]:
    data, _, _ = load_and_hash_json(path, "behavior evidence")

    evidence_hash = str(data.get("implementation_hash") or "").lower()
    if evidence_hash != implementation_hash.lower():
        fail("behavior evidence implementation_hash does not match computed implementation hash")

    checks = data.get("checks")
    if not isinstance(checks, dict):
        fail("behavior evidence must contain a checks object")

    missing = [
        check
        for check in sorted(REQUIRED_BEHAVIOR_CHECKS)
        if checks.get(check) is not True
    ]

    if missing:
        fail("behavior evidence missing required passing checks: " + ", ".join(missing))

    for flag in (
        "production_authorization",
        "deployment_performed",
        "observation_performed",
        "optimization_performed",
        "commit_performed",
        "push_performed",
    ):
        value = data.get(flag)
        if value is False:
            continue
        if str(value).strip().upper() in {"FALSE", "NOT_PERFORMED", "NOT_AUTHORIZED"}:
            continue
        fail(f"behavior evidence must declare {flag}=false")

    return data


def redact(obj: Any) -> Any:
    if isinstance(obj, dict):
        return {
            k: redact(v)
            for k, v in obj.items()
            if not any(marker in str(k).lower() for marker in SECRET_KEY_MARKERS)
        }
    if isinstance(obj, list):
        return [redact(item) for item in obj]
    return obj


def write_canonical_json(path: Path, obj: Any) -> None:
    path.parent.mkdir(parents=True, exist_ok=True)
    path.write_text(canonical_json(obj), encoding="utf-8")


def ensure_not_overwrite_upstream(
    outputs: Dict[str, Path],
    inputs: List[str],
    root: Path,
) -> None:
    input_paths = set()
    for item in inputs:
        if item:
            input_paths.add((root / normalize_ref(item)).resolve())

    for output_path in outputs.values():
        if output_path.resolve() in input_paths:
            fail(f"output would overwrite an upstream artifact: {output_path}")

def write_artifacts(
    outputs: Dict[str, Path],
    state: Dict[str, Any],
    evidence: Dict[str, Any],
    handoff: Dict[str, Any],
    manifest: Dict[str, Any],
    report: str,
) -> None:
    implementation_id = state["implementation_id"]
    implementation_hash = state["implementation_hash"]

    write_canonical_json(outputs["implementation_manifest"], manifest)
    write_canonical_json(outputs["evidence"], evidence)
    write_canonical_json(outputs["handoff"], handoff)

    manifest_json = canonical_json(manifest)
    manifest_literal = json.dumps(manifest_json)

    implementation_py = "\n".join(
        [
            '"""Deterministic D25 implementation manifest."""',
            "import json",
            "",
            f"IMPLEMENTATION_ID = {json.dumps(implementation_id)}",
            f"IMPLEMENTATION_HASH = {json.dumps(implementation_hash)}",
            "",
            f"MANIFEST_JSON = {manifest_literal}",
            "MANIFEST = json.loads(MANIFEST_JSON)",
            "",
            "",
            "def main() -> None:",
            "    print(json.dumps(MANIFEST, sort_keys=True, indent=2))",
            "",
            "",
            'if __name__ == "__main__":',
            "    main()",
            "",
        ]
    )

    outputs["implementation_py"].parent.mkdir(parents=True, exist_ok=True)
    outputs["implementation_py"].write_text(implementation_py, encoding="utf-8")

    markdown = "\n".join(
        [
            "# VS1 Implementation D25",
            "",
            "This document is generated by the Constitutional Implementation Actuator.",
            "",
            "```text",
            report,
            "```",
            "",
        ]
    )

    outputs["markdown"].parent.mkdir(parents=True, exist_ok=True)
    outputs["markdown"].write_text(markdown, encoding="utf-8")


def run_d25(args: argparse.Namespace) -> int:
    root = Path(args.root).resolve()
    firewall = ActionFirewall()

    input_paths = [
        args.objective_source,
        args.d22,
        args.d23,
        args.d24,
        args.isr,
        args.parent_implementation,
    ]
    if args.behavior_evidence:
        input_paths.append(args.behavior_evidence)

    try:
        firewall.authorize("verify_upstream")

        objective_source = verify_objective_source(root, args.objective_source)
        d22 = verify_d22(root, args.d22, objective_source)
        d23 = verify_d23(root, args.d23)
        d24 = verify_d24(root, args.d24, d23)
        isr = verify_isr(root, args.isr)
        parent = verify_parent(root, args.parent_implementation)

        coverage = validate_architecture(d24["architecture"])

        plan = compile_implementation_plan(
            d24=d24,
            architecture=d24["architecture"],
            d22=d22,
            d23=d23,
            objective_source=objective_source,
            isr=isr,
            firewall=firewall,
        )

        implementation_hash, implementation_id = compute_implementation_identity(
            plan=plan,
            d22=d22,
            d23=d23,
            d24=d24,
            objective_source=objective_source,
            isr=isr,
        )

        if implementation_hash == parent["parent_implementation_hash"]:
            fail("implementation hash is not distinct from parent implementation hash")
        if implementation_id == parent["parent_implementation_id"]:
            fail("implementation id is not distinct from parent implementation id")

        if args.plan_only:
            print(
                canonical_json(
                    {
                        "implementation_id": implementation_id,
                        "implementation_hash": implementation_hash,
                        "plan": plan,
                    }
                )
            )
            return 0

        if not args.behavior_evidence:
            fail(
                "behavior evidence is required for D25 PASS; "
                "run backend implementation tests and provide --behavior-evidence"
            )

        behavior = verify_behavior_evidence(
            root / normalize_ref(args.behavior_evidence),
            implementation_hash,
        )

        firewall.authorize("emit_evidence")

        outputs = {
            "implementation_manifest": root
            / args.out_dir
            / "implementation_d25_manifest.json",
            "implementation_py": root / args.out_dir / "implementation_d25.py",
            "evidence": root / args.out_dir / "implementation_d25_evidence.json",
            "handoff": root / args.out_dir / "deployment_handoff_d25.json",
            "markdown": root / args.docs_dir / "VS1_IMPLEMENTATION_D25.md",
        }

        ensure_not_overwrite_upstream(outputs, input_paths, root)

        verification = {
            "D25": f"{len(REQUIRED_BEHAVIOR_CHECKS)}/{len(REQUIRED_BEHAVIOR_CHECKS)}",
            "tests_vs1": behavior.get("summary", {}).get("tests_vs1", "NOT_PROVIDED"),
            "adjacent_suites": behavior.get("summary", {}).get("adjacent_suites", "NOT_PROVIDED"),
            "full_regression": behavior.get("summary", {}).get("full_regression", "NOT_COMPLETE"),
        }

        state = {
            "objective_source": objective_source,
            "d22": d22,
            "d23": d23,
            "d24": d24,
            "isr": isr,
            "parent": parent,
            "coverage": coverage,
            "plan": plan,
            "implementation_id": implementation_id,
            "implementation_hash": implementation_hash,
            "behavior": behavior,
            "verification": verification,
            "artifacts": {name: str(path) for name, path in outputs.items()},
        }

        lineage = {
            "implementation_id": implementation_id,
            "parent_implementation_id": parent["parent_implementation_id"],
            "selected_architecture_id": d24["candidate_id"],
            "selected_architecture_hash": d24["candidate_hash"],
            "selection_hash": d24["selection_hash"],
            "candidate_id": d24["candidate_id"],
            "candidate_hash": d24["candidate_hash"],
            "objective_id": EXPECTED_OBJECTIVE_ID,
            "objective_hash": d22["objective_hash"],
            "objective_source_ref": objective_source["source_ref"],
            "objective_source_sha256": objective_source["source_sha256"],
            "isr_hash": isr["isr_sha256"],
            "component_mappings": plan["component_mappings"],
            "requirement_mappings": {
                capability: [EXPECTED_OBJECTIVE_ID]
                for capability in sorted(REQUIRED_CAPABILITIES)
            },
            "security_mappings": {
                control: ["D24.architecture.security"]
                for control in sorted(REQUIRED_SECURITY)
            },
        }

        evidence = redact(
            {
                "gate": "D25",
                "policy_version": POLICY_VERSION,
                "status": "PASS",
                "production_authorization": False,
                "deployment_performed": False,
                "observation_performed": False,
                "optimization_performed": False,
                "commit_performed": False,
                "push_performed": False,
                "upstream": {
                    "d22": d22,
                    "d23": d23,
                    "d24": {
                        "d24_hash": d24["d24_hash"],
                        "candidate_id": d24["candidate_id"],
                        "candidate_hash": d24["candidate_hash"],
                        "selection_hash": d24["selection_hash"],
                        "candidate_hash_verified_by": d24["candidate_hash_verified_by"],
                    },
                    "isr": isr,
                    "objective_source": objective_source,
                },
                "implementation": {
                    "implementation_id": implementation_id,
                    "implementation_hash": implementation_hash,
                    "backend_id": plan["backend_id"],
                    "parent_implementation_id": parent["parent_implementation_id"],
                },
                "coverage": coverage,
                "lineage": lineage,
                "behavior": {
                    "checks": behavior.get("checks", {}),
                    "summary": behavior.get("summary", {}),
                },
                "firewall_actions": sorted(firewall.actions),
                "artifacts": state["artifacts"],
            }
        )

        handoff = redact(
            {
                "implementation_id": implementation_id,
                "implementation_hash": implementation_hash,
                "selected_architecture_id": d24["candidate_id"],
                "selected_architecture_hash": d24["candidate_hash"],
                "selection_hash": d24["selection_hash"],
                "candidate_id": d24["candidate_id"],
                "candidate_hash": d24["candidate_hash"],
                "objective_id": EXPECTED_OBJECTIVE_ID,
                "objective_hash": d22["objective_hash"],
                "isr_hash": isr["isr_sha256"],
                "backend_id": plan["backend_id"],
                "parent_implementation_id": parent["parent_implementation_id"],
                "verification_summary": verification,
                "production_authorization": False,
                "deployment_performed": False,
            }
        )

        manifest = redact(
            {
                "implementation_id": implementation_id,
                "implementation_hash": implementation_hash,
                "plan": plan,
                "lineage": lineage,
                "policy_version": POLICY_VERSION,
            }
        )

        report = render_stop_report(state)

        if args.write:
            write_artifacts(
                outputs=outputs,
                state=state,
                evidence=evidence,
                handoff=handoff,
                manifest=manifest,
                report=report,
            )

        print(report)
        return 0

    except FailClosed as exc:
        print(render_blocked_report(str(exc)))
