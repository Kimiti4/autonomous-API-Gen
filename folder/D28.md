# 63. GOVERNING PRINCIPLE

The purpose of D28 is not to make the system appear successful.

The purpose is to establish:

```text
WHAT THE EVIDENCE ACTUALLY SHOWS
WHAT CAN REASONABLY BE INFERRED
WHAT REMAINS UNKNOWN
WHAT CONTRADICTS
WHAT CLAIMS ARE UNSUPPORTED
```

The authoritative chain is:

```text

    ↓
OBJECTIVE
    ↓
CANDIDATES
    ↓

    ↓
IMPLEMENTATION
    ↓
DEPLOYMENT ARTIFACT
    ↓

    ↓
```

D28 must preserve the distinction between:

```text

UNKNOWN
```

These are different classes of information and must never be collapsed.

The D28 interpretation is therefore an **epistemic artifact**, not an evolution decision.

The system must know what it knows, what it merely infers, what it does not know, and what evidence conflicts.

Only after that boundary is complete may a later authority decide what the system should do.

This is the **full gate**, rather than the earlier blueprint. It is intentionally strict about the most important architectural property here: **D27 produces facts, D28 produces epistemic meaning, and a later gate owns decisions**.

class FailClosed(Exception):
    pass


class ActionFirewall:
    def __init__(self) -> None:
        self.actions: List[str] = []

    def authorize(self, action: str) -> None:
        if action in PROHIBITED_ACTIONS:
            raise FailClosed(f"prohibited action requested: {action}")
        if action not in ALLOWED_ACTIONS:
            raise FailClosed(f"action not authorized by D28 firewall: {action}")
        self.actions.append(action)


def canonical_json(obj: Any) -> str:
    return json.dumps(
        obj,
        sort_keys=True,
        separators=(",", ":"),
        ensure_ascii=False,
    )


def canonical_bytes(obj: Any) -> bytes:
    return canonical_json(obj).encode("utf-8")


def sha256_bytes(data: bytes) -> str:
    return hashlib.sha256(data).hexdigest()


def sha256_obj(obj: Any) -> str:
    return sha256_bytes(canonical_bytes(obj))


def sha256_file(path: Path) -> str:
    return hashlib.sha256(path.read_bytes()).hexdigest()


def fail(message: str) -> None:
    raise FailClosed(message)


def load_json(path: Path, label: str) -> Dict[str, Any]:
    if not path.is_file():
        fail(f"{label} missing: {path}")

    try:
        data = json.loads(path.read_text(encoding="utf-8"))
    except Exception as exc:
        fail(f"{label} is not valid JSON: {path}: {exc}")

    if not isinstance(data, dict):
        fail(f"{label} must contain a JSON object: {path}")

    return data


def scan_for_secrets(obj: Any) -> List[str]:
    text = canonical_json(obj)
    findings: List[str] = []

    for name, pattern in SECRET_PATTERNS:
        if pattern.search(text):
            findings.append(name)

    return sorted(set(findings))


def verify_evidence_file(
    evidence_path: Path,
    expected_hash: Optional[str],
) -> Dict[str, Any]:
    if not evidence_path.is_file():
        fail(f"D27 evidence missing: {evidence_path}")

    raw_hash = sha256_file(evidence_path).lower()

    evidence = load_json(evidence_path, "D27 evidence")

    embedded_hash = str(evidence.get("evidence_hash") or "").lower()

    if expected_hash:
        expected_norm = expected_hash.lower()
        if raw_hash != expected_norm and embedded_hash != expected_norm:
            fail("D27 evidence hash does not match expected authoritative hash")
    else:
        if not embedded_hash.startswith(EXPECTED_D27_EVIDENCE_HASH_PREFIX):
            fail(
                "D27 evidence hash cannot be verified; "
                "expected hash missing and embedded hash does not match known prefix"
            )

    secret_findings = scan_for_secrets(evidence)
    if secret_findings:
        fail(
            "secret-shaped material detected in D27 evidence: "
            + ", ".join(secret_findings)
        )

    return evidence


def verify_provenance(evidence: Dict[str, Any]) -> Dict[str, Any]:
    required_fields = [
        "run_id",
        "deployment_id",
        "implementation_id",
        "candidate_id",
        "selection_id",
        "candidate_generation_id",
        "objective_id",
        "isr_hash",
        "observations",
    ]

    for field in required_fields:
        if field not in evidence:
            fail(f"D27 evidence missing required provenance field: {field}")

    if evidence["run_id"] != EXPECTED_D27_RUN_ID:
        fail("D27 run_id mismatch")

    if evidence["deployment_id"] != EXPECTED_DEPLOYMENT_ID:
        fail("D26 deployment identity mismatch")

    if evidence["implementation_id"] != EXPECTED_IMPLEMENTATION_ID:
        fail("D25 implementation identity mismatch")

    if evidence["candidate_id"] != EXPECTED_CANDIDATE_ID:
        fail("D24 selected candidate identity mismatch")

    if evidence["selection_id"] != EXPECTED_SELECTION_ID:
        fail("D24 selection identity mismatch")

    if evidence["candidate_generation_id"] != EXPECTED_CANDIDATE_GENERATION_ID:
        fail("D23 candidate generation identity mismatch")

    if evidence["objective_id"] != EXPECTED_OBJECTIVE_ID:
        fail("D22 objective identity mismatch")

    if evidence["isr_hash"] != EXPECTED_ISR_HASH:
        fail("ISR hash mismatch")

    return {
        "d27_run_id": evidence["run_id"],
        "deployment_id": evidence["deployment_id"],
        "implementation_id": evidence["implementation_id"],
        "candidate_id": evidence["candidate_id"],
        "selection_id": evidence["selection_id"],
        "candidate_generation_id": evidence["candidate_generation_id"],
        "objective_id": evidence["objective_id"],
        "isr_hash": evidence["isr_hash"],
    }


def normalize_observation(obs: Dict[str, Any]) -> Dict[str, Any]:
    required = ["observation_id", "semantic_key", "result"]
    for field in required:
        if field not in obs:
            fail(f"observation missing required field: {field}")

    tags = set(obs.get("tags", []))
    if not isinstance(tags, set):
        tags = set(tags)

    result = str(obs.get("result") or "").strip().lower()
    if result not in {"success", "failure", "unknown"}:
        fail(f"observation result invalid: {obs.get('observation_id')}")

    if result == "success":
        tags.add("success")
    elif result == "failure":
        tags.add("failure")
    else:
        tags.add("unknown")

    semantic = {
        "observation_id": obs["observation_id"],
        "semantic_key": obs["semantic_key"],
        "category": obs.get("category", "unclassified"),
        "result": result,
        "priority": obs.get("priority"),
        "http_status": obs.get("http_status"),
        "tags": sorted(tags),
        "scope": sorted(obs.get("scope", [])),
    }

    return semantic


def normalize_observations(evidence: Dict[str, Any]) -> List[Dict[str, Any]]:
    raw_observations = evidence.get("observations", [])

    if not isinstance(raw_observations, list):
        fail("D27 observations must be a list")

    if not raw_observations:
        fail("D27 observations missing")

    normalized = [normalize_observation(obs) for obs in raw_observations]

    # Reject duplicate observation identities with conflicting semantics.
    by_id: Dict[str, Dict[str, Any]] = {}
    for obs in normalized:
        obs_id = obs["observation_id"]
        if obs_id in by_id:
            if canonical_json(by_id[obs_id]) != canonical_json(obs):
                fail(f"duplicate observation identity with conflicting values: {obs_id}")
        else:
            by_id[obs_id] = obs

    normalized_sorted = sorted(normalized, key=lambda item: item["observation_id"])
    return normalized_sorted


def detect_contradictions(
    observations: List[Dict[str, Any]],
) -> List[Dict[str, Any]]:
    by_semantic_key: Dict[str, List[Dict[str, Any]]] = {}

    for obs in observations:
        by_semantic_key.setdefault(obs["semantic_key"], []).append(obs)

    contradictions: List[Dict[str, Any]] = []
    contradiction_index = 0

    for semantic_key, items in sorted(by_semantic_key.items()):
        if len(items) < 2:
            continue

        canonical_values = {canonical_json(item) for item in items}
        if len(canonical_values) > 1:
            contradiction_index += 1
            contradictions.append(
                {
                    "contradiction_id": f"CONTRA-{contradiction_index:03d}",
                    "semantic_key": semantic_key,
                    "left_evidence_id": items[0]["observation_id"],
                    "right_evidence_id": items[1]["observation_id"],
                    "field": "semantic_payload",
                    "left_value": items[0]["result"],
                    "right_value": items[1]["result"],
                    "identity_context": semantic_key,
                    "classification": "factual",
                    "impact": "affected claims downgraded; positive conclusions blocked",
                    "resolution_status": "UNRESOLVED",
                }
            )

    return contradictions


def observation_has_tags(obs: Dict[str, Any], tags: Iterable[str]) -> bool:
    obs_tags = set(obs.get("tags", []))
    return all(tag in obs_tags for tag in tags)


def evidence_strength_for_observations(
    observations: List[Dict[str, Any]],
) -> str:
    if not observations:
        return "INSUFFICIENT"

    obs_ids = {obs["observation_id"] for obs in observations}
    if len(obs_ids) > 1:
        return "CORROBORATED"

    return "DIRECT"


def evaluate_criteria(
    policy: Dict[str, Any],
    observations: List[Dict[str, Any]],
    contradictions: List[Dict[str, Any]],
) -> tuple[List[Dict[str, Any]], List[Dict[str, Any]], List[Dict[str, Any]]]:
    criteria = policy.get("criteria", [])
    if not isinstance(criteria, list) or not criteria:
        fail("D28 policy must define criteria")

    contradiction_keys = {
        contradiction["semantic_key"] for contradiction in contradictions
    }

    requirement_coverage: List[Dict[str, Any]] = []
    claims: List[Dict[str, Any]] = []
    unknowns: List[Dict[str, Any]] = []

    unknown_index = 0

    for criterion in criteria:
        criterion_id = criterion.get("id")
        criterion_text = criterion.get("text", "")
        required_tag_sets = criterion.get("required_tag_sets", [])

        if not criterion_id:
            fail("criterion missing id")
        if not isinstance(required_tag_sets, list):
            fail(f"criterion {criterion_id} required_tag_sets must be a list")

        supporting: List[Dict[str, Any]] = []
        satisfied = True

        for tag_set in required_tag_sets:
            if not isinstance(tag_set, list):
                fail(f"criterion {criterion_id} contains malformed tag set")

            matches = [
                obs
                for obs in observations
                if observation_has_tags(obs, tag_set)
            ]

            if matches:
                supporting.extend(matches)
            else:
                satisfied = False

        supporting_sorted = sorted(
            supporting,
            key=lambda item: item["observation_id"],
        )

        supporting_ids = [obs["observation_id"] for obs in supporting_sorted]
        has_contradiction = any(
            obs["semantic_key"] in contradiction_keys for obs in supporting_sorted
        )

        if has_contradiction:
            status = "CONTRADICTION"
            strength = "CONTRADICTED"
        elif satisfied:
            status = "OBSERVED"
            strength = evidence_strength_for_observations(supporting_sorted)
        elif supporting_sorted:
            status = "UNKNOWN"
            strength = "LIMITED"
        else:
            status = "UNKNOWN"
            strength = "INSUFFICIENT"

        scope = [
            "loopback-fixture",
            "exercised-workload",
            "non-production",
        ]

        limitations = []

        if status == "UNKNOWN":
            limitations.append(
                "Evidence is insufficient to establish this criterion for the exercised workload."
            )
            unknown_index += 1
            unknowns.append(
                {
                    "unknown_id": f"UNKNOWN-{unknown_index:03d}",
                    "question": f"Can {criterion_id} be established from authoritative D27 evidence?",
                    "why_unknown": "Required observations are missing, incomplete, or not semantically sufficient.",
                    "missing_evidence": required_tag_sets,
                    "affected_requirement_or_obligation": [criterion_id],
                    "risk_if_assumed": "Assuming success would convert missing evidence into an unsupported claim.",
                    "possible_future_evidence": [
                        "A future authorized D27 run with observations satisfying the required semantic tags."
                    ],
                }
            )

        if status == "CONTRADICTION":
            limitations.append(
                "Authoritative observations conflict for this criterion."
            )

        requirement_coverage.append(
            {
                "criterion_id": criterion_id,
                "criterion_text": criterion_text,
                "status": status,
                "supporting_observations": supporting_ids,
                "evidence_strength": strength,
                "scope": scope,
                "limitations": limitations,
            }
        )

        if status in {"OBSERVED", "CONTRADICTION"}:
            claim_id = f"CLAIM-{criterion_id}"
            claims.append(
                {
                    "claim_id": claim_id,
                    "statement": (
                        f"{criterion_text} was directly observed within the exercised D27 loopback workload."
                        if status == "OBSERVED"
                        else f"{criterion_text} is affected by contradictory D27 evidence."
                    ),
                    "epistemic_class": status,
                    "evidence_strength": strength,
                    "supporting_observations": supporting_ids,
                    "scope": scope,
                    "limitations": limitations,
                }
            )

    return requirement_coverage, claims, unknowns


def evaluate_inferences(
    policy: Dict[str, Any],
    observations: List[Dict[str, Any]],
) -> List[Dict[str, Any]]:
    inferences = policy.get("inferences", [])
    if not isinstance(inferences, list):
        fail("D28 policy inferences must be a list")

    claims: List[Dict[str, Any]] = []

    for inference in inferences:
        claim_id = inference.get("claim_id")
        statement = inference.get("statement")
        required_tag_sets = inference.get("required_tag_sets", [])
        reasoning = inference.get("reasoning")
        confidence = inference.get("confidence", "LIMITED")
        scope = inference.get("scope", ["loopback-fixture", "exercised-workload"])

        if not claim_id or not statement or not reasoning:
            fail("inference rule missing required fields")

        if not isinstance(required_tag_sets, list) or not required_tag_sets:
            fail(f"inference {claim_id} must define required_tag_sets")

        supporting: List[Dict[str, Any]] = []
        satisfied = True

        for tag_set in required_tag_sets:
            matches = [
                obs
                for obs in observations
                if observation_has_tags(obs, tag_set)
            ]
            if matches:
                supporting.extend(matches)
            else:
                satisfied = False

        if not satisfied:
            continue

        supporting_ids = sorted({obs["observation_id"] for obs in supporting})

        if not supporting_ids:
            fail(f"inference {claim_id} has no supporting observations")

        claims.append(
            {
                "claim_id": claim_id,
                "statement": statement,
                "epistemic_class": "INFERRED",
                "evidence_strength": confidence,
                "supporting_observations": supporting_ids,
                "scope": scope,
                "limitations": [
                    "This is an inference, not a direct runtime observation.",
                    "It applies only within the declared scope.",
                ],
                "reasoning": reasoning,
            }
        )

    return claims


def validate_claim_scope(
    claim: Dict[str, Any],
    policy: Dict[str, Any],
) -> None:
    forbidden_scopes = set(policy.get("forbidden_scopes", []))
    claim_scopes = set(claim.get("scope", []))

    invalid = forbidden_scopes.intersection(claim_scopes)
    if invalid:
        fail(
            f"claim {claim.get('claim_id')} attempts unsupported scope: "
            + ", ".join(sorted(invalid))
        )

def canonical_interpretation_payload(
    evidence: Dict[str, Any],
    policy: Dict[str, Any],
    provenance: Dict[str, Any],
    requirement_coverage: List[Dict[str, Any]],
    claims: List[Dict[str, Any]],
    contradictions: List[Dict[str, Any]],
    unknowns: List[Dict[str, Any]],
) -> Dict[str, Any]:
    payload = {
        "interpretation_version": INTERPRETATION_VERSION,
        "policy_version": policy.get("policy_version", POLICY_VERSION),
        "objective_id": provenance["objective_id"],
        "source_evidence_id": provenance["d27_run_id"],
        "source_evidence_hash": evidence.get("evidence_hash", ""),
        "d27_run_id": provenance["d27_run_id"],
        "deployment_id": provenance["deployment_id"],
        "implementation_id": provenance["implementation_id"],
        "candidate_id": provenance["candidate_id"],
        "selection_id": provenance["selection_id"],
        "candidate_generation_id": provenance["candidate_generation_id"],
        "isr_hash": provenance["isr_hash"],
        "claims": sorted(claims, key=lambda item: item["claim_id"]),
        "requirement_coverage": sorted(
            requirement_coverage,
            key=lambda item: item["criterion_id"],
        ),
        "contradictions": sorted(
            contradictions,
            key=lambda item: item["contradiction_id"],
        ),
        "unknowns": sorted(unknowns, key=lambda item: item["unknown_id"]),
        "unsupported_claims": [],
        "provenance": build_provenance(evidence, provenance),
        "canonicalization": {
            "version": CANONICALIZATION_VERSION,
            "excluded_nonsemantic_fields": sorted(NON_SEMANTIC_OBSERVATION_FIELDS),
            "observation_ordering": "observation_id",
        },
        "scope": sorted(evidence.get("scope", [])),
        "limitations": DEFAULT_LIMITATIONS,
    }

    return payload

def run_d28(args: argparse.Namespace) -> int:
    firewall = ActionFirewall()

    try:
        firewall.authorize("verify_evidence")

        evidence_path = Path(args.d27_evidence)
        evidence = verify_evidence_file(evidence_path, args.expected_d27_hash)
        provenance = verify_provenance(evidence)

        policy = load_json(Path(args.policy), "D28 policy")

        firewall.authorize("normalize_evidence")

        observations = normalize_observations(evidence)
        contradictions = detect_contradictions(observations)

        firewall.authorize("interpret_evidence")

        requirement_coverage, criterion_claims, unknowns = evaluate_criteria(
            policy,
            observations,
            contradictions,
        )

        inference_claims = evaluate_inferences(policy, observations)
        claims = criterion_claims + inference_claims

        for claim in claims:
            validate_claim_scope(claim, policy)

        payload = canonical_interpretation_payload(
            evidence=evidence,
            policy=policy,
            provenance=provenance,
            requirement_coverage=requirement_coverage,
            claims=claims,
            contradictions=contradictions,
            unknowns=unknowns,
        )

        interpretation_hash = sha256_obj(payload).lower()
        interpretation_id = f"vs1-d28-interp-{interpretation_hash[:16]}"

        interpretation = dict(payload)
        interpretation["interpretation_id"] = interpretation_id
        interpretation["interpretation_hash"] = interpretation_hash

        counts = {
            "observed": len(
                [claim for claim in claims if claim["epistemic_class"] == "OBSERVED"]
            ),
            "inferred": len(
                [claim for claim in claims if claim["epistemic_class"] == "INFERRED"]
            ),
            "unknown": len(unknowns),
            "contradictions": len(contradictions),
            "unsupported_claims": len(interpretation["unsupported_claims"]),
        }

        coverage_summary = {
            item["criterion_id"]: item["status"]
            for item in requirement_coverage
        }

        artifacts = [
            "folder/VS1_RUNTIME_INTERPRETATION_D28.md",
            "vertical_slice/runtime_interpretation_d28.py",
            "vertical_slice/runtime_interpretation_d28_evidence.json",
            "vertical_slice/runtime_interpretation_d28_manifest.json",
            "vertical_slice/runtime_interpretation_d28_schema.json",
            "vertical_slice/runtime_interpretation_d28_handoff.json",
            "tests/vs1/test_d28_runtime_interpretation.py",
        ]

        state = {
            "interpretation": interpretation,
            "counts": counts,
            "coverage_summary": coverage_summary,
            "artifacts": artifacts,
        }

        report = render_pass_report(state)

        if args.write:
            out_dir = Path(args.out_dir)
            docs_dir = Path(args.docs_dir)

            out_dir.mkdir(parents=True, exist_ok=True)
            docs_dir.mkdir(parents=True, exist_ok=True)

            (out_dir / "runtime_interpretation_d28_evidence.json").write_text(
                canonical_json(interpretation),
                encoding="utf-8",
            )

            handoff = {
                "d28_status": "PASS",
                "interpretation_id": interpretation_id,
                "interpretation_hash": interpretation_hash,
                "source_d27_run": provenance["d27_run_id"],
                "source_d27_evidence_hash": interpretation["source_evidence_hash"],
                "deployment_id": provenance["deployment_id"],
                "implementation_id": provenance["implementation_id"],
                "candidate_id": provenance["candidate_id"],
                "objective_id": provenance["objective_id"],
                "isr_hash": provenance["isr_hash"],
                "claim_count": len(claims),
                "unknown_count": len(unknowns),
                "contradiction_count": len(contradictions),
                "unsupported_claim_count": len(interpretation["unsupported_claims"]),
                "requirement_coverage_summary": coverage_summary,
                "security_summary": "bounded_to_exercised_workload",
                "operational_summary": "mechanical_observations_only",
                "scope": interpretation["scope"],
                "limitations": interpretation["limitations"],
                "next_gate": "D29",
                "authorization_state": "D29_AUTHORIZATION=NOT GRANTED",
            }

            (out_dir / "runtime_interpretation_d28_handoff.json").write_text(
                canonical_json(handoff),
                encoding="utf-8",
            )

            (docs_dir / "VS1_RUNTIME_INTERPRETATION_D28.md").write_text(
                "\n".join(
                    [

                        "",

                        "```",
                        "",
                    ]
                ),
                encoding="utf-8",
            )

        print(report)
        return 0

    except FailClosed as exc:
        print(render_fail_report(str(exc)))
        return 1
